# Yunshu Multidimensional Intelligent Document 系统初始化示例

// 初始化系统
const yycSystem = new YYCSystem({
  storage: {
    type: 'indexedDB',
    name: 'yyc-documents'
  },
  ai: {
    apiKey: 'your-ai-key'
  }
});

// 使用日期组件
const currentDate = yycSystem.modules['date-time'].insertDynamicDate('full');
console.log('当前日期:', currentDate);

// 创建思维导图
const mindMap = yycSystem.modules.diagrams.createMindMap({
  root: '云枢文档系统',
  children: ['核心功能', '扩展功能', '协作系统']
});

// 执行Python代码
yycSystem.modules['code-processor'].executeCode(
  'print(2+2)',
  'python'
).then(result => console.log('执行结果:', result));
